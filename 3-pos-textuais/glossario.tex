% Termos técnicos empregados no trabalho. As siglas são gerenciadas
% separadamente em 1-pre-textuais/lista-de-abreviaturas-e-siglas.tex.

\newglossaryentry{esbeltez}{
    name={esbeltez},
    description={característica geométrica associada à relação entre o comprimento efetivo de uma barra comprimida e seu raio de giração, utilizada na avaliação de estabilidade}
}

\newglossaryentry{flambagem}{
    name={flambagem},
    description={fenômeno de instabilidade no qual um elemento submetido à compressão pode apresentar deslocamentos laterais significativos}
}

\newglossaryentry{parser}{
    name={parser},
    description={componente de software que interpreta dados estruturados em determinado formato e os transforma em uma representação interna utilizável pelo programa}
}

\newglossaryentry{pre-dimensionamento}{
    name={pré-dimensionamento},
    description={estimativa inicial da composição e das dimensões de elementos estruturais com base em solicitações, propriedades dos materiais e critérios de verificação}
}

\newglossaryentry{secao-composta}{
    name={seção composta},
    description={seção transversal formada pela associação de dois ou mais componentes, tratada segundo hipóteses específicas de interação entre eles}
}

\newglossaryentry{topologia-estrutural}{
    name={topologia estrutural},
    description={organização dos nós e dos elementos de uma estrutura, incluindo suas relações de conectividade}
}

\newglossaryentry{web-worker}{
    name={Web Worker},
    description={recurso de navegadores que permite executar tarefas em uma linha de execução separada da interface principal}
}
